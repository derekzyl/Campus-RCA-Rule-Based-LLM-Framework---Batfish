~OSPF & ACL CONFIGURATIONS~

###Standard ACL FOR MGT###

configure terminal

access-list 1 permit 192.168.10.0 0.0.0.255 
access-list 1 deny any

line vty 0 15

access-class 1 in

exec-timeout 20 0

end
write memory

##Guest WLAN in 11.10.65.0/24 should not reach internal network####
configure terminal

ip access-list extended GUEST-WLAN-FILTER

 remark BLOCK GUEST ACCESS TO WIRED CAMPUS LAN
 deny ip 11.10.65.0 0.0.0.255 192.168.0.0 0.0.255.255

 remark BLOCK GUEST ACCESS TO STAFF WLAN
 deny ip 11.10.65.0 0.0.0.255 11.10.55.0 0.0.0.255

 remark BLOCK GUEST ACCESS TO STUDENT WLAN
 deny ip 11.10.65.0 0.0.0.255 11.10.60.0 0.0.0.255

 remark ALLOW REMAINING TRAFFIC
 permit ip 11.10.65.0 0.0.0.255 any

exit

##Apply inbound on VLAN 65##
interface vlan 65
 ip access-group GUEST-WLAN-FILTER in
exit

end
write memory

##DMZ Policy, network is 12.20.20.0/26##

#Inside → DMZ          Allowed as required

access-list DMZ-IN extended deny ip 12.20.20.0 255.255.255.192 11.10.0.0 255.255.0.0

##CONFIGURATION ON BOTH CORE SWITCHES##
configure terminal

ip access-list extended STUDENT-FILTER

 remark BLOCK MANAGEMENT NETWORK
 deny ip 192.168.30.0 0.0.0.255 192.168.10.0 0.0.0.255

 remark BLOCK NETWORK SERVICES EXCEPT DNS
 permit udp 192.168.30.0 0.0.0.255 host 192.168.80.13 eq domain
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.80.13 eq domain

 deny ip 192.168.30.0 0.0.0.255 192.168.80.0 0.0.0.255

 remark ALLOW OTHER TRAFFIC
 permit ip 192.168.30.0 0.0.0.255 any

exit

interface vlan 30
 ip access-group STUDENT-FILTER in
exit

end
